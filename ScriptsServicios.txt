CREATE FUNCTION `EmpleadoRegistra` (rfcEmp VARCHAR(15))
RETURNS VARCHAR(256) 
BEGIN
DECLARE rfcResult VARCHAR(256);
SELECT CONCAT(nombre,' ',apellidoP,' ',apellidoM) INTO rfcResult
FROM empleado_Ap
WHERE rfc = rfcEmp;
RETURN rfcResult;
END;

CREATE FUNCTION `getOrdenesPago` (idSolicitud INT)
RETURNS INT
BEGIN
DECLARE idOrden INT;
SELECT op.idOrdenesPago INTO idOrden
FROM ordenes_pago op, ordenes_pago_has_solicitud ophs
WHERE op.idOrdenesPago = ophs.OrdenesPago_idOrdenPago
AND ophs.Solicitud_idSolicitud = idSolicitud;
RETURN idOrden;
END;
//

DELIMITER ;


DELIMITER //
CREATE FUNCTION `getCalculoActuaria` (idSolicitud INT)
RETURNS INT
BEGIN
DECLARE idCalculo INT;
SELECT ca.idCalculoActuaria into idCalculo  
FROM calculo_actuaria ca, calculo_actuaria_has_solicitud cahs
WHERE ca.idCalculoActuaria = cahs.CalculoActuaria_idCalculo 
and cahs.Solicitud_idSolicitud  = idSolicitud;
RETURN idCalculo;
END;
//

DELIMITER ;
